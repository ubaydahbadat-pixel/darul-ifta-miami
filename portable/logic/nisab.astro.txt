---
import Base from '../layouts/Base.astro';
import { nisab, fitr } from '../data/site';

const usd = (n: number) => n.toLocaleString('en-US', { style: 'currency', currency: 'USD' });
const ratio = nisab.goldPerOz / nisab.silverPerOz;
const updated = new Date(nisab.date + 'T12:00:00Z').toLocaleDateString('en-US', { month: 'long', day: 'numeric', year: 'numeric', timeZone: 'UTC' });
const auto = nisab.source !== 'Entered by hand';
// Mahr (Fatāwā Dār al-ʿUlūm Zakariyyā, vol. 3, Kitāb al-Nikāḥ, Bāb 4): dirhams of silver, at 3.0618 g per dirham
// (10 dirhams = 30.618 g; 480 = 1,469.664 g; 500 = 1,530.900 g, as given there).
const DIRHAM_G = 3.0618;
const mahr = [
  { name: 'Minimum mahr', dirhams: 10 },
  { name: 'Mahr Fāṭimī', dirhams: 480 },
  { name: 'Mahr al-Azwāj', dirhams: 500 },
].map((m) => {
  const grams = m.dirhams * DIRHAM_G;
  return { ...m, grams, value: (grams / 31.1034768) * nisab.silverPerOz };
});
// The full texts of the two fatāwā behind the mahr figures, shown when the reader opens the source.
const grams = (g: number) => g.toLocaleString('en-US', { minimumFractionDigits: 3, maximumFractionDigits: 3 });
const time = new Date(nisab.updatedAt).toLocaleTimeString('en-US', { hour: 'numeric', minute: '2-digit', timeZone: 'America/New_York' }).toLowerCase();
---
